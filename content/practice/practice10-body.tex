\underline{Exercise 10.5}
\begin{enumerate}
% Question 1
\item[1.] Find the derivatives of:
\begin{multicols}{3}
\begin{enumerate}
\item[(a)] $y=e^{2t+4}$
\item[(b)] $y=e^{1-9t}$
\item[(c)] $y=e^{t^{2}+1}$
\item[(d)] $y=5e^{2-t^{2}}$
\item[(e)] $y=e^{ax^{2}+bx+c}$
\item[(f)] $y=xe^{x}$
\item[(g)] $y=x^{2}e^{2x}$
\item[(h)] $y=axe^{bx+c}$
\end{enumerate}
\end{multicols}

% Question 3
\item[3.] Find the derivatives of:
\begin{multicols}{3}
\begin{enumerate}
\item[(a)] $y=\ln\left(7t^{5}\right)$
\item[(b)] $y=\ln\left(at^{10}\right)$
\item[(c)] $y=\ln(t+19)$
\item[(d)] $y=5\ln(t+1)^{2}$
\item[(e)] $y=\ln x-\ln(1+x)$
\item[(f)] $y=\ln\left[x(1-x)^{8}\right]$
\item[(g)] $y=\ln\left(\frac{2x}{1+x}\right)$
\item[(h)] $y=5x^{4}\ln x^{2}$
\end{enumerate}
\end{multicols}

% Question 7
\item[7.] Find the derivatives of the following by first taking the natural log of both sides:
\begin{multicols}{2}
\begin{enumerate}
\item[(a)] $y=\dfrac{3x}{(x+2)(x+4)}$
\item[(b)] $y=\left(x^{2}+3\right)e^{x^{2}+1}$
\end{enumerate}
\end{multicols}
\end{enumerate}
