\end{enumerate}

% Question 7
\item[7.]
\begin{enumerate}
\item[(a)] Since $\ln y=\ln 3x-\ln(x+2)-\ln(x+4)$,
\[ \frac{1}{y}\frac{dy}{dx}=\frac{1}{x}-\frac{1}{x+2}-\frac{1}{x+4}=\frac{8-x^{2}}{x(x+2)(x+4)} \]
Hence,
\[ \frac{dy}{dx}=\frac{8-x^{2}}{x(x+2)(x+4)}\cdot\frac{3x}{(x+2)(x+4)}=\frac{3\left(8-x^{2}\right)}{(x+2)^{2}(x+4)^{2}} \]
\item[(b)] Since $\ln y=\ln\left(x^{2}+3\right)+x^{2}+1$,
\[ \frac{1}{y}\frac{dy}{dx}=\frac{2x}{x^{2}+3}+2x=\frac{2x\left(x^{2}+4\right)}{x^{2}+3} \]
Hence,
\[ \frac{dy}{dx}=\frac{2x\left(x^{2}+4\right)}{x^{2}+3}\cdot\left(x^{2}+3\right)e^{x^{2}+1}=2x\left(x^{2}+4\right)e^{x^{2}+1} \]
\end{enumerate}
\end{enumerate}
