\underline{Exercise 8.5}
\begin{enumerate}
% Question 1
\item[1.]
\begin{enumerate}
\item[(a)] With $F_{x}=-6$ and $F_{y}=1$, the implicit-function rule gives
\[\frac{dy}{dx}=-\frac{F_{x}}{F_{y}}=-\frac{-6}{1}=6\]
Equivalently, solving for $y=6x-7$ and differentiating gives $dy/dx=6$.
\item[(b)] With $F_{x}=12$ and $F_{y}=3$,
\[\frac{dy}{dx}=-\frac{F_{x}}{F_{y}}=-\frac{12}{3}=-4\]

\item[(c)] With $F_{x}=2x+6$ and $F_{y}=-1$,
\[\frac{dy}{dx}=-\frac{F_{x}}{F_{y}}=-\frac{2x+6}{-1}=2x+6\]

\end{enumerate}

\item[3.]
\begin{enumerate}
\item[(a)] 
\[F_{x}=2xy^{3}+yz \qquad F_{y}=3x^{2}y^{2}+xz \qquad F_{z}=2z+xy\]
\[\frac{\partial y}{\partial x}=-\frac{F_{x}}{F_{y}}=-\frac{2xy^{3}+yz}{3x^{2}y^{2}+xz}\]
\[\frac{\partial y}{\partial z}=-\frac{F_{z}}{F_{y}}=-\frac{2z+xy}{3x^{2}y^{2}+xz}\]

\end{enumerate}
