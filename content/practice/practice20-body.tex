\underline{Additional Examples}
\begin{enumerate}
% Question 1
\item[1.] Utility maximization. Consider the problem
\[\max_{\{x_1, x_2\}} \quad x_1^{\alpha} x_2^{\beta} \quad \text{s.t.} \quad p_1 x_1 + p_2 x_2 = m\]
where $p_1$ and $p_2$ are the prices of goods 1 and 2, and $m$ is the income available to spend on the two goods. Find the optimal $x_1^*$ and $x_2^*$.

% Question 2
\item[2.] Cost minimization. A firm chooses capital $K$ and labor $L$ to produce a fixed quantity $\bar{Q}$ at the lowest cost, where $p$ is the price of capital and $w$ is the price of labor:
\[\min_{\{K, L\}} \quad pK + wL \quad \text{s.t.} \quad Q(K, L) = \bar{Q}\]
Set up the Lagrange function, write down the first-order conditions, and find the two conditions that the optimal $K^*$ and $L^*$ satisfy.

% Question 3
\item[3.] Intertemporal utility maximization. Your utility from consumption $c_1$ in period 1 and $c_2$ in period 2 is
\[U(c_1, c_2) = \ln c_1 + \beta \ln c_2 \qquad 0 < \beta < 1\]
You earn income $y_1 > 0$ in period 1 and $y_2 > 0$ in period 2, and any income $s$ you save in period 1 earns interest $r > 0$, so $c_1 + s = y_1$ and $c_2 = y_2 + (1 + r)s$. Combining these, your budget constraint is
\[c_1 + \frac{1}{1+r} c_2 = m, \qquad \text{where } m = y_1 + \frac{1}{1+r} y_2\]
is your present-discounted income.
\begin{enumerate}
\item[(a)] Write down the Lagrange function and the first-order conditions.
\item[(b)] Find the optimal $c_1^*$, $c_2^*$, and $\lambda^*$.
\item[(c)] How does the maximum utility $U^*$ change when present-discounted income $m$ increases slightly?
\end{enumerate}

\end{enumerate}
