% Question 2
\item[2.] $y=5 x^{2}-4 x$
\begin{enumerate}
\item[(a)]
\[
\begin{aligned}
\frac{\Delta y}{\Delta x} &=\frac{f(x+\Delta x)-f(x)}{\Delta x} \\
&=\frac{5(x+\Delta x)^{2}-4(x+\Delta x)-\left(5 x^{2}-4 x\right)}{\Delta x} \\
&=\frac{5\left(x^{2}+2 x \Delta x+(\Delta x)^{2}\right)-4 x-4 \Delta x-5 x^{2}+4 x}{\Delta x} \\
&=\frac{10 x \Delta x+5 (\Delta x)^{2}-4 \Delta x}{\Delta x}=10 x-4+5 \Delta x
\end{aligned}
\]
\item[(b)] \[\frac{d y}{d x}=\lim _{\Delta x \rightarrow 0} \frac{\Delta y}{\Delta x}=\lim _{\Delta x \rightarrow 0}\,(10 x-4+5 \Delta x)=10 x-4\]
\item[(c)] $f^{\prime}(2)=10 \times 2-4=16$ and $f^{\prime}(3)=10 \times 3-4=26$.
\end{enumerate}

% Question 3
\item[3.] $y=5 x-2$
\begin{enumerate}
\item[(a)] \[\frac{\Delta y}{\Delta x}=\frac{5(x+\Delta x)-2-(5 x-2)}{\Delta x}=\frac{5 \Delta x}{\Delta x}=5\]
It is a constant function.
\item[(b)] No: the difference quotient equals 5 whatever the size of $\Delta x$, so its limit as $\Delta x \rightarrow 0$ is also 5, i.e.\ $d y / d x=5$.
\end{enumerate}
\end{enumerate}
